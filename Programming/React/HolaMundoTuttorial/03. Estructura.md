[+] Si se esta corriendo desde un devcontainer hay que asegurarse de poner esto en el vite.config:

```js
import { defineConfig } from 'vite'

import react from '@vitejs/plugin-react'

// https://vitejs.dev/config/

export default defineConfig({
  plugins: [react()],
  server: {
    host: '0.0.0.0',  // Allows external access
    port: 5173,       // Ensures the same port is used
    strictPort: true  // Prevents fallback to a different port if 5173 is in use
  }
});
```
