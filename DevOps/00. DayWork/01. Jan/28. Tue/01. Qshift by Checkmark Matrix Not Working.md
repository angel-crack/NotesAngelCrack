function updateTeamSchedules(engineerName, inputQDays) {
    // Convert inputQDays into the desired grouped format
    const groupedDays = inputQDays.reduce((acc, item) => {
        const [queue, day] = item.split('-');
        const group = acc.find(([key]) => key === queue);
        if (group) {
            group[1].push(day);
        } else {
            acc.push([queue, [day]]);
        }
        return acc;
    }, []);

    // Iterate through profile.teamSchedules.queues to update engineers
    groupedDays.forEach(([queue, days]) => {
        // Find the matching queue in the profile
        const queueObj = profile.teamSchedules.queues.find(q => q.queue === queue);
        if (queueObj) {
            // Iterate through the days in the queue and update engineers
            queueObj.days.forEach(dayObj => {
                if (days.includes(dayObj.day)) {
                    // Add engineerName to the engineers array if it's not already there
                    if (!dayObj.engineers.includes(engineerName)) {
                        dayObj.engineers.push(engineerName);
                    }
                }
            });
        }
    });
}
